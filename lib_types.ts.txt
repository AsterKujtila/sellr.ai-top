export type Tab = 'chat' | 'offers' | 'store' | 'credits' | 'history';

export interface UserProfile {
  id: string;
  name: string;
  email: string;
  avatar_url?: string;
  whop_store_url?: string;
  plan: 'free' | 'starter' | 'pro' | 'scale';
  credits_used: number;
  credits_limit: number;
  onboarding_completed: boolean;
  created_at: string;
}

export interface ChatSession {
  id: string;
  user_id: string;
  title: string;
  session_date: string;
  day_of_week: string;
  summary?: string;
  created_at: string;
  updated_at: string;
}

export interface ChatMessage {
  id: string;
  session_id: string;
  user_id: string;
  role: 'user' | 'assistant';
  content: string;
  created_at: string;
}

export interface Feedback {
  id: string;
  user_id: string;
  reason: string;
  message: string;
  created_at: string;
}

export interface ShellContextType {
  activeTab: Tab;
  setActiveTab: (tab: Tab) => void;
  settingsOpen: boolean;
  setSettingsOpen: (open: boolean) => void;
  chatActive: boolean;
  setChatActive: (active: boolean) => void;
}
