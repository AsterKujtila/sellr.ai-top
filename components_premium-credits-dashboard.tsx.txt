'use client';

import { useEffect, useState } from 'react';
import { Zap, TrendingUp, Award, Star, ChevronRight } from 'lucide-react';
import { motion } from 'framer-motion';
import { createClient } from '@/lib/supabase/client';

interface PricingPlan {
  tier: string;
  price: number;
  credits: number;
  features: string[];
  popular?: boolean;
}

const PRICING_PLANS: PricingPlan[] = [
  {
    tier: 'Starter',
    price: 19,
    credits: 200,
    features: [
      'Chat coaching (200 credits)',
      'Basic analytics',
      'Email support',
    ],
  },
  {
    tier: 'Pro',
    price: 49,
    credits: 600,
    features: [
      'Chat coaching (600 credits)',
      'Offer copy generator',
      'Store analyzer',
      'Priority support',
      'Advanced analytics',
    ],
    popular: true,
  },
  {
    tier: 'Scale',
    price: 99,
    credits: 1500,
    features: [
      'Unlimited chat coaching',
      'All tools included',
      'VIP support',
      'Custom integrations',
      'White-label analytics',
    ],
  },
];

export default function PremiumCreditsDashboard() {
  const [profile, setProfile] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const loadProfile = async () => {
      try {
        const supabase = createClient();
        const { data: { user } } = await supabase.auth.getUser();
        if (user) {
          const { data } = await supabase
            .from('profiles')
            .select('*')
            .eq('id', user.id)
            .single();
          setProfile(data);
        }
      } catch (err) {
        console.error('Failed to load profile:', err);
      } finally {
        setLoading(false);
      }
    };
    loadProfile();
  }, []);

  if (loading) {
    return (
      <div className="flex-center h-full">
        <div className="text-center">
          <div className="w-8 h-8 border-2 border-primary/30 border-t-primary rounded-full animate-spin mx-auto mb-4" />
          <p className="text-sm text-muted-foreground">Loading dashboard...</p>
        </div>
      </div>
    );
  }

  const remaining = profile?.credits_limit - profile?.credits_used || 0;
  const percentUsed = ((profile?.credits_used || 0) / (profile?.credits_limit || 1)) * 100;

  return (
    <div className="flex flex-col h-full bg-gradient-dark overflow-y-auto no-scrollbar">
      {/* Header */}
      <div className="px-4 pt-4 pb-6 space-y-4">
        <h2 className="font-serif text-2xl font-bold text-gradient">Credits</h2>

        {/* Credits Card */}
        <motion.div
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          className="card rounded-2xl space-y-4"
        >
          <div className="flex items-start justify-between">
            <div>
              <p className="text-sm text-muted-foreground mb-2">Credits Available</p>
              <div className="space-y-2">
                <p className="font-serif text-4xl font-bold text-gradient">{remaining}</p>
                <p className="text-xs text-muted-foreground">{profile?.plan} plan</p>
              </div>
            </div>
            <Zap className="w-12 h-12 text-warning/20" />
          </div>

          {/* Progress Bar */}
          <div className="space-y-2">
            <div className="flex justify-between items-center">
              <span className="text-10px text-muted-foreground uppercase">Used</span>
              <span className="text-10px text-muted-foreground">
                {profile?.credits_used}/{profile?.credits_limit}
              </span>
            </div>
            <div className="w-full h-2 bg-surface-variant rounded-full overflow-hidden">
              <motion.div
                initial={{ width: 0 }}
                animate={{ width: `${Math.min(percentUsed, 100)}%` }}
                transition={{ duration: 1, delay: 0.3 }}
                className="h-full bg-gradient-to-r from-primary to-secondary"
              />
            </div>
          </div>
        </motion.div>

        {/* Cost Breakdown */}
        <motion.div
          initial={{ opacity: 0, y: 10 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ delay: 0.1 }}
          className="card rounded-xl"
        >
          <p className="text-sm font-semibold mb-4">Feature Costs</p>
          <div className="grid grid-cols-3 gap-3">
            {[
              { icon: '💬', label: 'Chat', cost: '1 cr' },
              { icon: '✍️', label: 'Offers', cost: '2 cr' },
              { icon: '🏪', label: 'Store', cost: '3 cr' },
            ].map(({ icon, label, cost }) => (
              <div key={label} className="flex flex-col items-center p-3 bg-surface-variant rounded-lg">
                <span className="text-2xl mb-2">{icon}</span>
                <p className="text-xs font-medium">{label}</p>
                <p className="text-10px text-muted-foreground mt-1">{cost}</p>
              </div>
            ))}
          </div>
        </motion.div>
      </div>

      {/* Pricing Plans */}
      <div className="px-4 pb-8 space-y-4">
        <div>
          <h3 className="font-semibold text-foreground mb-4">Upgrade Your Plan</h3>

          <div className="space-y-3">
            {PRICING_PLANS.map((plan, i) => (
              <motion.div
                key={plan.tier}
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: i * 0.1 }}
                className={cn(
                  'card rounded-2xl cursor-pointer group',
                  plan.popular ? 'ring-2 ring-primary border-primary' : ''
                )}
              >
                {plan.popular && (
                  <div className="absolute -top-3 left-4">
                    <span className="bg-primary text-white text-10px font-bold px-3 py-1 rounded-full flex items-center gap-1">
                      <Star className="w-3 h-3" />
                      MOST POPULAR
                    </span>
                  </div>
                )}

                <div className="flex items-start justify-between mb-4">
                  <div>
                    <p className="font-semibold text-foreground">{plan.tier}</p>
                    <p className="flex items-baseline gap-1 mt-2">
                      <span className="font-serif text-2xl font-bold text-primary">${plan.price}</span>
                      <span className="text-10px text-muted-foreground">/month</span>
                    </p>
                  </div>
                  <Award className="w-6 h-6 text-primary/40 group-hover:text-primary transition-colors" />
                </div>

                <p className="text-sm text-muted-foreground mb-4">{plan.credits} credits/month</p>

                <ul className="space-y-2 mb-4">
                  {plan.features.map((feature) => (
                    <li key={feature} className="flex gap-2 items-start text-sm">
                      <ChevronRight className="w-4 h-4 text-primary flex-shrink-0 mt-0.5" />
                      <span className="text-muted-foreground">{feature}</span>
                    </li>
                  ))}
                </ul>

                <motion.button
                  whileHover={{ scale: 1.02 }}
                  whileTap={{ scale: 0.98 }}
                  className={cn(
                    'w-full py-3 rounded-lg font-semibold transition-all',
                    plan.popular
                      ? 'btn-primary'
                      : 'btn-secondary'
                  )}
                  onClick={() => window.open('https://whop.com/sellr-ai', '_blank')}
                >
                  Upgrade Now
                </motion.button>
              </motion.div>
            ))}
          </div>

          <p className="text-10px text-muted-foreground text-center mt-6">
            💳 Secure payments via Whop • Credits never expire
          </p>
        </div>
      </div>
    </div>
  );
}

function cn(...classes: any[]) {
  return classes.filter(Boolean).join(' ');
}
